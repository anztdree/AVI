   173→    fun createStt() {

   176→            override fun onListeningStarted() {
   177→                if (testCallback == null) setState(AviState.LISTENING)
   178→                _partialText.value = ""
   179→            }
   180→
   181→            override fun onRms(level: Float) {
   182→                _rmsLevel.value = level
   183→            }
   184→
   185→            override fun onPartial(text: String) {
   186→                _partialText.value = text
   187→            }
   188→
   189→            override fun onResult(text: String) {
   190→                _partialText.value = ""
   191→                val cb = testCallback
   192→                if (cb != null) {
   193→                    testCallback = null
   194→                    _state.value = AviState.IDLE
   195→                    cb(text)
   196→                } else {
   197→                    handleUserInput(text)
   198→                }
   199→            }
   200→
   201→            override fun onError(message: String) {
   202→                _partialText.value = ""
   203→                if (testCallback != null) {
   204→                    testCallback = null
   205→                    _state.value = AviState.IDLE
   206→                    _error.value = message
   207→                } else {
   208→                    if (_state.value == AviState.LISTENING) setState(AviState.IDLE)
   209→                    _error.value = message
   210→                    _conversationActive.value = false
   211→                }
   212→            }
   213→        })
   214→    }

   273→    /** Aksi instan lokal: protokol, buka aplikasi, kondisi HP/jam/tanggal */
   274→    private fun handleLocal(text: String): Boolean {
   275→        val ctx = getApplication<Application>()
   276→
   277→        protocolRepo.matchPhrase(text)?.let {
   278→            _messages.value = _messages.value + ChatMessage(role = "user", text = text)
   279→            runProtocol(it)
   280→            return true
   281→        }
   282→
   283→        val lower = text.lowercase(Locale.ROOT).trim()
   284→        if (lower.startsWith("buka ") || lower.startsWith("open ") ||
   285→            (lower.contains(" buka ") && lower.length < 60)
   286→        ) {
   287→            val query = text.replace(Regex("(?i)(tolong|coba|sekarang|ya|dunk|dong|pak|bu|aplikasi|app|open|buka)"), " ")
   288→                .replace(Regex("\\s+"), " ").trim()
   289→            if (query.isNotEmpty()) {
   290→                val apps = DeviceActions.findApps(ctx, query)
   291→                if (apps.isNotEmpty()) {
   292→                    val app = apps.first()
   293→                    DeviceActions.openApp(ctx, app.packageName)
   294→                    addLocalReply("Siap pak, ${app.label} sudah saya buka.")
   295→                } else {
   296→                    addLocalReply("Tidak ada aplikasi yang cocok dengan \"$query\" di HP, pak.")
   297→                }
   298→                return true
   299→            }
   300→        }
   301→
   302→        DeviceActions.localAnswer(text, _diagnostics.value ?: DeviceActions.readDiagnostics(ctx))
   303→            ?.let { addLocalReply(it); return true }
   304→
   305→        return false
   306→    }

   308→    private fun addLocalReply(reply: String) {
   309→        _messages.value = _messages.value + ChatMessage(role = "avi", text = reply)
   310→        persist()
   311→        if (_settings.value.autoSpeak && _settings.value.autoListen) {
   312→            setState(AviState.SPEAKING)
   313→            tts.speak(cleanForSpeech(reply))
   314→        } else {
   315→            maybeAutoListen()
   316→        }
   317→    }

   319→    // ===== Alur ke model AI =====
   320→
   321→    private fun sendToModel(userText: String) {
   322→        if (_state.value == AviState.THINKING || _state.value == AviState.LIVE) return
   323→        _error.value = null
   324→        _messages.value = _messages.value + ChatMessage(role = "user", text = userText)
   325→        setState(AviState.THINKING)
   326→        _streamingText.value = ""
   327→
   328→        val runtime = _settings.value.copy(
   329→            memoryText = memoryRepo.joinedText(),
   330→            recap = buildRecap(),
   331→            researchMode = isResearchRequest(userText)
   332→        )
   333→
   334→        val history = historyRepo.contextWindow(_messages.value)
   335→        val client = AiClientFactory.create(runtime)
   336→
   337→        val callback = object : AiStreamCallback {
   338→            override fun onDelta(text: String) {
   339→                _streamingText.value += text
   340→            }
   341→
   342→            override fun onComplete(fullText: String) {
   343→                viewModelScope.launch {
   344→                    settingsRepo.bumpUsage()
   345→                    finishReply(fullText)
   346→                }
   347→            }
   348→
   349→            override fun onError(message: String) {
   350→                setState(AviState.IDLE)
   351→                _streamingText.value = ""
   352→                _error.value = message
   353→                _conversationActive.value = false
   354→                pendingSteps.clear()
   355→            }
   356→        }
   357→
   358→        streamJob = viewModelScope.launch {
   359→            try {
   360→                client.stream(history, runtime, callback)
   361→            } catch (e: CancellationException) {
   362→                throw e
   363→            } catch (e: Exception) {
   364→                setState(AviState.IDLE)
   365→                _streamingText.value = ""
   366→                _error.value = "Gagal terhubung ke server AI: ${e.message ?: "unknown"}"
   367→                _conversationActive.value = false
   368→                pendingSteps.clear()
   369→            }
   370→        }
   371→    }

   373→    private fun isResearchRequest(text: String): Boolean {
   374→        val t = text.lowercase(Locale.ROOT)
   375→        return t.startsWith("riset") || t.contains("risetkan") || t.contains("riset tentang") ||
   376→            t.contains("teliti tentang") || t.contains("cari tahu tentang")
   377→    }

   379→    /** X7 — kesinambungan topik: potongan obrolan sebelumnya (pesan lama) */
   380→    private fun buildRecap(): String {
   381→        val today = SimpleDateFormat("yyyyMMdd", Locale.US).format(Date())
   382→        val old = _messages.value.filter {
   383→            SimpleDateFormat("yyyyMMdd", Locale.US).format(Date(it.timestamp)) != today
   384→        }
   385→        if (old.isEmpty()) return ""
   386→        val tail = old.takeLast(6).joinToString(" ") {
   387→            (if (it.role == "user") "Pemilik: " else "AVI: ") + it.text
   388→        }
   389→        return tail.take(500)
   390→    }

   392→    private fun finishReply(raw: String) {
   393→        val ctx = getApplication<Application>()
   394→
   395→        // [INGAT: ...] → memori jangka panjang
   396→        Regex("\\[INGAT\\s*:\\s*(.*?)]", setOf(RegexOption.IGNORE_CASE, RegexOption.DOT_MATCHES_ALL))
   397→            .findAll(raw)
   398→            .forEach { m ->
   399→                val fact = m.groupValues[1].trim()
   400→                if (fact.isNotEmpty()) {
   401→                    viewModelScope.launch {
   402→                        memoryRepo.add(fact)
   403→                        _memories.value = memoryRepo.current
   404→                    }
   405→                }
   406→            }
   407→
   408→        // [BUKA: ...] → aksi buka aplikasi dari model
   409→        Regex("\\[BUKA\\s*:\\s*(.*?)]", setOf(RegexOption.IGNORE_CASE, RegexOption.DOT_MATCHES_ALL))
   410→            .findAll(raw)
   411→            .forEach { m ->
   412→                val apps = DeviceActions.findApps(ctx, m.groupValues[1])
   413→                if (apps.isNotEmpty()) DeviceActions.openApp(ctx, apps.first().packageName)
   414→            }
   415→
   416→        var text = raw
   417→            .replace(Regex("\\[(INGAT|BUKA|LUPAKAN|PROTOKOL|GAMBAR)\\s*:.*?]", setOf(RegexOption.IGNORE_CASE, RegexOption.DOT_MATCHES_ALL)), " ")
   418→            .replace(Regex("\\[/?[A-Z_]*]"), " ")
   419→            .replace(Regex("\\s+"), " ")
   420→            .trim()
   421→        if (text.isBlank()) text = "Siap pak."
   422→
   423→        _messages.value = _messages.value + ChatMessage(role = "avi", text = text)
   424→        _streamingText.value = ""
   425→        persist()
   426→        speakReply(text)
   427→    }

   431→    private fun speakReply(text: String) {
   432→        val speech = cleanForSpeech(text)
   433→        val s = _settings.value
   434→        if (!s.autoSpeak || speech.isBlank()) {
   435→            afterSpeak()
   436→            return
   437→        }
   438→        setState(AviState.SPEAKING)
   439→        if (s.provider == com.anztdree.avi.data.AiProvider.GEMINI &&
   440→            s.cloudTtsEnabled && s.cloudTtsModel.isNotBlank()
   441→        ) {
   442→            viewModelScope.launch {
   443→                CloudTts.synthesize(speech, s)
   444→                    .onSuccess { cloudPlayer.playPcm(it.pcm, it.sampleRate) }
   445→                    .onFailure { tts.speak(speech) } // fallback suara Android
   446→            }
   447→        } else {
   448→            tts.speak(speech)
   449→        }
   450→    }

   469→    private fun afterSpeak() {

   472→            processQueue()
   473→            return
   474→        }
   475→        maybeAutoListen()

   481→                delay(600)
   482→                if (_conversationActive.value && _state.value == AviState.IDLE) {
   483→                    stt?.start()
   484→                }
   485→            }
   486→        }

   489→    private fun greetingText(): String {

   492→            in 4..10 -> "Selamat pagi"
   493→            in 11..14 -> "Selamat siang"

   502→    fun runProtocol(p: Protocol) {
   503→        if (p.steps.isEmpty()) {
   504→            _error.value = "Protokol \"${p.name}\" belum punya langkah, pak."
   505→            return
   506→        }
   507→        _messages.value = _messages.value + ChatMessage(role = "user", text = "▶ ${p.name} dijalankan")
   508→        pendingSteps.clear()
   509→        pendingSteps.addAll(p.steps)
   510→        processQueue()
   511→    }

   513→    private fun processQueue() {
   514→        if (_state.value != AviState.IDLE) return
   515→        val next = pendingSteps.removeFirstOrNull() ?: return
   516→        sendToModel(next)
   517→    }

   519→    /** X4 — briefing pagi proaktif */
   520→    private fun maybeBriefing(s: AviSettings) {
   521→        viewModelScope.launch {
   522→            val configured = when (s.provider) {
   523→                com.anztdree.avi.data.AiProvider.GEMINI ->
   524→                    s.geminiKey.isNotBlank() && s.geminiModel.isNotBlank()
   525→                else -> s.ocKey.isNotBlank()
   526→            }
   527→            if (!configured || settingsRepo.isBriefedToday()) return@launch
   528→            val hour = Calendar.getInstance().get(Calendar.HOUR_OF_DAY)
   529→            if (hour in 5..10) {
   530→                settingsRepo.markBriefed()
   531→                delay(2500)
   532→                _messages.value = _messages.value + ChatMessage(
   533→                    role = "avi",
   534→                    text = "Selamat pagi, ${s.userName}. AVI siapkan briefing pagi Anda…"
   535→                )
   536→                pendingSteps.add(
   537→                    "Briefing pagi singkat pakai pencarian web: cuaca hari ini, lalu 2 berita utama " +
   538→                        "Indonesia hari ini. Jawaban maksimal 4 kalimat, tutup dengan satu kalimat semangat."
   539→                )
   540→                processQueue()
   541→            }
   542→        }
   543→    }

   547→    fun startLive() {
   548→        val s = _settings.value
   549→        if (liveWs != null || _state.value == AviState.LIVE) return
   550→        if (s.provider != com.anztdree.avi.data.AiProvider.GEMINI || s.geminiKey.isBlank()) {
   551→            _error.value = "Live mode butuh API key Gemini, pak."
   552→            return
   553→        }
   554→        if (_models.value.isEmpty()) {
   555→            _error.value = "Muat daftar model dulu di Pusat Kendali, pak — AVI perlu tahu model live terbaru."
   556→            return
   557→        }
   558→        val model = ModelCatalog.pickLiveModel(_models.value, s.geminiModel)
   559→        if (model == null) {
   560→            _error.value = "Tidak ada model live di daftar terbaru, pak. Coba muat ulang daftar model."
   561→            return
   562→        }
   563→        _conversationActive.value = false
   564→        setState(AviState.LIVE)
   565→        liveTranscript.clear()
   566→        val prompt = SystemPrompts.build(s.copy(memoryText = memoryRepo.joinedText()))
   567→        liveWs = LiveClient(s.geminiKey, model, prompt, object : LiveClient.Listener {
   568→            override fun onSetupComplete() {}
   569→
   570→            override fun onAudioPcm(pcm: ByteArray) {
   571→                livePlayer.enqueue(pcm, 24000)
   572→            }
   573→
   574→            override fun onTranscript(text: String) {
   575→                liveTranscript.append(text)
   576→            }
   577→
   578→            override fun onInterrupted() {
   579→                livePlayer.stop()
   580→            }
   581→
   582→            override fun onTurnComplete() {
   583→                val t = liveTranscript.toString().trim()
   584→                liveTranscript.clear()
   585→                if (t.isNotEmpty()) {
   586→                    _messages.value = _messages.value + ChatMessage(role = "avi", text = t)
   587→                    persist()
   588→                }
   589→            }
   590→
   591→            override fun onError(message: String) {
   592→                stopLive()
   593→                _error.value = message
   594→            }
   595→
   596→            override fun onClosed() {
   597→                if (_state.value == AviState.LIVE) stopLive()
   598→            }
   599→        }).also { it.connect() }
   600→
   601→        mic.start(
   602→            onChunk = { pcm, level ->
   603→                _rmsLevel.value = level
   604→                liveWs?.sendAudioPcm(pcm)
   605→            },
   606→            onError = { msg ->
   607→                stopLive()
   608→                _error.value = msg
   609→            }
   610→        )
   611→    }

   615→        liveWs?.close()
   616→        liveWs = null
   617→        livePlayer.stop()
   618→        _voiceLevel.value = 0f
   619→        if (_state.value == AviState.LIVE) setState(AviState.IDLE)
   620→    }

   735→    private fun setState(s: AviState) {
   736→        _state.value = s
   737→        val pattern = when (s) {
   738→            AviState.LISTENING -> longArrayOf(0, 40)
   739→            AviState.THINKING -> longArrayOf(0, 60, 80, 60)
   740→            AviState.SPEAKING -> longArrayOf(0, 30, 60, 30)
   741→            AviState.LIVE -> longArrayOf(0, 80, 60, 80)
   742→            AviState.IDLE -> return
   743→        }
   744→        if (!_settings.value.hapticsEnabled || _settings.value.powerSave) return
   745→        try {
   746→            val v = getApplication<Application>().getSystemService(Context.VIBRATOR_SERVICE) as? Vibrator
   747→            if (v != null && v.hasVibrator()) {
   748→                if (Build.VERSION.SDK_INT >= 26) {
   749→                    v.vibrate(VibrationEffect.createWaveform(pattern, -1))
   750→                } else {
   751→                    @Suppress("DEPRECATION") v.vibrate(pattern, -1)
   752→                }
   753→            }
   754→        } catch (_: Exception) {
   755→        }
   756→    }
